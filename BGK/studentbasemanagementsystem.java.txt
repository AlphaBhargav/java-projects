import javax.swing.*;
import java.awt.event.*;
import java.io.*;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.LinkedHashMap;
import java.util.Map;

public class StudentManagement extends JFrame implements ActionListener {

    JLabel l1, l2, l3;
    JTextField t1, t2, t3;
    JButton b1, b2, b3, b4, b5;

    String logFileName = "student_operations.log";
    String dataFileName = "students_data.txt";
    Map<String, String[]> students = new LinkedHashMap<>();

    StudentManagement() {
        initializeStorage();
        logAction("Application started");

        if (java.awt.GraphicsEnvironment.isHeadless()) {
            logAction("Headless environment detected; UI not displayed.");
            return;
        }

        setTitle("Student Management System");
        setLayout(null);
        setSize(550, 300);

        l1 = new JLabel("Student ID");
        l2 = new JLabel("Name");
        l3 = new JLabel("Course");

        t1 = new JTextField();
        t2 = new JTextField();
        t3 = new JTextField();

        b1 = new JButton("Add");
        b2 = new JButton("Search");
        b3 = new JButton("Update");
        b4 = new JButton("Delete");
        b5 = new JButton("Clear");

        l1.setBounds(30, 30, 100, 25);
        t1.setBounds(130, 30, 150, 25);

        l2.setBounds(30, 70, 100, 25);
        t2.setBounds(130, 70, 150, 25);

        l3.setBounds(30, 110, 100, 25);
        t3.setBounds(130, 110, 150, 25);

        b1.setBounds(20, 180, 80, 30);
        b2.setBounds(110, 180, 90, 30);
        b3.setBounds(210, 180, 90, 30);
        b4.setBounds(310, 180, 90, 30);
        b5.setBounds(410, 180, 90, 30);

        add(l1);
        add(t1);
        add(l2);
        add(t2);
        add(l3);
        add(t3);
        add(b1);
        add(b2);
        add(b3);
        add(b4);
        add(b5);

        b1.addActionListener(this);
        b2.addActionListener(this);
        b3.addActionListener(this);
        b4.addActionListener(this);
        b5.addActionListener(this);

        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setVisible(true);
    }

    private void initializeStorage() {
        loadStudentsFromFile();
    }

    private void loadStudentsFromFile() {
        File file = new File(dataFileName);
        if (!file.exists()) {
            return;
        }

        try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
            String line;
            boolean firstLine = true;
            while ((line = reader.readLine()) != null) {
                if (firstLine) {
                    firstLine = false;
                    continue;
                }
                if (line.trim().isEmpty()) {
                    continue;
                }
                String[] parts = line.split(",", 3);
                if (parts.length == 3) {
                    students.put(parts[0], new String[]{parts[1], parts[2]});
                }
            }
        } catch (IOException ex) {
            logAction("Load data file failed: " + ex.getMessage());
        }
    }

    private void saveStudentsToFile() {
        try (PrintWriter out = new PrintWriter(new FileWriter(dataFileName, false))) {
            out.println("id,name,course");
            for (Map.Entry<String, String[]> entry : students.entrySet()) {
                out.println(entry.getKey() + "," + entry.getValue()[0] + "," + entry.getValue()[1]);
            }
        } catch (IOException ex) {
            logAction("Data file save failed: " + ex.getMessage());
        }
    }

    private void logAction(String message) {
        String timestamp = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss").format(new Date());
        try (PrintWriter out = new PrintWriter(new FileWriter(logFileName, true))) {
            out.println(timestamp + " - " + message);
        } catch (IOException ex) {
            System.out.println("Log write failed: " + ex.getMessage());
        }
    }

    private void clearFields() {
        t1.setText("");
        t2.setText("");
        t3.setText("");
    }

    public void actionPerformed(ActionEvent e) {

        if (e.getSource() == b1) {
            String id = t1.getText().trim();
            String name = t2.getText().trim();
            String course = t3.getText().trim();

            if (id.isEmpty() || name.isEmpty() || course.isEmpty()) {
                JOptionPane.showMessageDialog(this, "Please fill all fields");
                return;
            }

            students.put(id, new String[]{name, course});
            saveStudentsToFile();
            logAction("Student added successfully. ID: " + id);
            JOptionPane.showMessageDialog(this, "Student Added Successfully");
        }

        if (e.getSource() == b2) {
            String sid = JOptionPane.showInputDialog("Enter Student ID");
            if (sid == null) {
                return;
            }

            String[] data = students.get(sid.trim());
            if (data != null) {
                t1.setText(sid.trim());
                t2.setText(data[0]);
                t3.setText(data[1]);
                logAction("Student found during search. ID: " + sid.trim());
            } else {
                clearFields();
                logAction("Student not found during search. ID: " + sid.trim());
                JOptionPane.showMessageDialog(this, "Student Not Found");
            }
        }

        if (e.getSource() == b3) {
            String id = t1.getText().trim();
            if (students.containsKey(id)) {
                students.put(id, new String[]{t2.getText().trim(), t3.getText().trim()});
                saveStudentsToFile();
                logAction("Student updated successfully. ID: " + id);
                JOptionPane.showMessageDialog(this, "Student Updated");
            } else {
                logAction("Update failed. Student not found. ID: " + id);
                JOptionPane.showMessageDialog(this, "Student Not Found");
            }
        }

        if (e.getSource() == b4) {
            String sid = JOptionPane.showInputDialog("Enter Student ID");
            if (sid == null) {
                return;
            }

            String key = sid.trim();
            if (students.remove(key) != null) {
                clearFields();
                saveStudentsToFile();
                logAction("Student deleted successfully. ID: " + key);
                JOptionPane.showMessageDialog(this, "Student Deleted");
            } else {
                logAction("Delete failed. Student not found. ID: " + key);
                JOptionPane.showMessageDialog(this, "Student Not Found");
            }
        }

        if (e.getSource() == b5) {
            clearFields();
            logAction("Input fields cleared");
        }
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(StudentManagement::new);
    }
}
